-- DRAGGING
-- ============================================================

local dragging = false
local dragInput = nil
local dragStart = nil
local startPosition = nil

local function updateDrag(input)
	if not dragging then
		return
	end

	local delta = input.Position - dragStart
	card.Position = UDim2.new(
		startPosition.X.Scale,
		startPosition.X.Offset + delta.X,
		startPosition.Y.Scale,
		startPosition.Y.Offset + delta.Y
	)
end

card.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPosition = card.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

card.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch then

		dragInput = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if input == dragInput then
		updateDrag(input)
	end
end)

-- ============================================================
-- MESSAGE POOL & SANITIZER
-- ============================================================

local messages = {}
local seen = {}

local function add(message)
	if type(message) ~= "string" then
		return
	end

	local trimmed = message:match("^%s*(.-)%s*$")
	if not trimmed or trimmed == "" then
		return
	end

	-- Reject standalone single letters (e.g. "a", "k", "z")
	if #trimmed == 1 and trimmed:match("%a") then
		return
	end

	-- Reject short 1-2 character ASCII fragments/gibberish
	if #trimmed <= 2 and trimmed:match("^%a+$") then
		return
	end

	if not seen[trimmed] then
		seen[trimmed] = true
		table.insert(messages, trimmed)
	end
end

-- ============================================================
-- RANDOM MESSAGE MEMORY
-- ============================================================

local recentMessages = {}
local RECENT_MEMORY = 80

local function wasRecentlyUsed(message)
	for i = 1, #recentMessages do
		if recentMessages[i] == message then
			return true
		end
	end
	return false
end

local function rememberMessage(message)
	table.insert(recentMessages, message)
	if #recentMessages > RECENT_MEMORY then
		table.remove(recentMessages, 1)
	end
end

-- ============================================================
-- SELECTION
-- ============================================================

local function getRandomMessage()
	if #messages == 0 then
		return "window"
	end

	for _ = 1, 50 do
		local candidate = messages[math.random(1, #messages)]
		if not wasRecentlyUsed(candidate) then
			rememberMessage(candidate)
			return candidate
		end
	end

	local candidate = messages[math.random(1, #messages)]
	rememberMessage(candidate)
	return candidate
end

-- ============================================================
-- CHAT DISPATCH
-- ============================================================

local function sendChat(message)
	pcall(function()
		local channels = TextChatService:FindFirstChild("TextChannels")
		if channels then
			local channel = channels:FindFirstChild("RBXGeneral")
			if channel then
				channel:SendAsync(message)
			end
		end
	end)

	pcall(function()
		local events = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
		if events then
			local request = events:FindFirstChild("SayMessageRequest")
			if request then
				request:FireServer(message, "All")
			end
		end
	end)
end

-- ============================================================
-- MOVEMENT
-- ============================================================

local function getHumanoid()
	local character = player.Character
	if not character then
		return nil
	end
	return character:FindFirstChildOfClass("Humanoid")
end

local function moveStick(dir)
	-- Movement relies on the TouchGui which only exists in PlayerGui
	local pg = player:FindFirstChild("PlayerGui")
	if not pg then return end

	local touchGui = pg:FindFirstChild("TouchGui")
	if not touchGui then return end

	local frame = touchGui:FindFirstChild("TouchControlFrame")
	if not frame then return end

	local dyn = frame:FindFirstChild("DynamicThumbstickFrame")
	if not dyn then return end

	local endImg = dyn:FindFirstChild("ThumbstickEnd")
	if not endImg then return end

	endImg.Visible = true
	endImg.AnchorPoint = Vector2.new(0.5, 0.5)

	local cx = dyn.AbsoluteSize.X * 0.5
	local cy = dyn.AbsoluteSize.Y * 0.55
	local offset = (dir == 1) and -65 or 65

	endImg.Position = UDim2.new(0, cx, 0, cy + offset)
end

-- ============================================================
-- STATE CONTROLLERS
-- ============================================================

local function stopAll()
	if walkConn then
		walkConn:Disconnect()
		walkConn = nil
	end

	if chatThread then
		task.cancel(chatThread)
		chatThread = nil
	end

	local humanoid = getHumanoid()
	if humanoid then
		humanoid:Move(Vector3.zero, true)
	end
end

local function startAll()
	stopAll()

	direction = 1
	nextSwitch = tick() + SWITCH_TIME
	nextJump = tick() + math.random(5, 9)

	walkConn = RunService.RenderStepped:Connect(function()
		local humanoid = getHumanoid()
		if not humanoid or humanoid.Health <= 0 then
			return
		end

		humanoid:Move(Vector3.new(0, 0, -direction), true)
		moveStick(direction)

		if tick() >= nextSwitch then
			direction = -direction
			nextSwitch = tick() + SWITCH_TIME
		end

		if tick() >= nextJump then
			humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
			nextJump = tick() + math.random(6, 12)
		end
	end)

	chatThread = task.spawn(function()
		task.wait(1)
		if enabled then
			sendChat(getRandomMessage())
		end

		while enabled do
			task.wait(math.random(367, 512))
			if enabled then
				sendChat(getRandomMessage())
			end
		end
	end)
end

		cardStroke.Color = Color3.fromRGB(65, 65, 72)
	end
end
